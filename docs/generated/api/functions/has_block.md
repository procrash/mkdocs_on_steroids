    // Create a piece block
    piece_block block(0, 0);

    // Create a has_block object with the piece block
    has_block hb(block);

    // Create a pending block with the same piece block
    pending_block pb{block, 1};

    // Use the function call operator to compare
    bool result = hb(pb);

    // Output the result

        // The blocks are equal

}
